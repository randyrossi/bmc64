cmake_minimum_required(VERSION 3.13)

if (NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif ()
if (NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to a Pico SDK checkout with TinyUSB initialized")
endif ()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")

project(mock_hid C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if (NOT PICO_BOARD STREQUAL "pico")
    message(FATAL_ERROR "This fixture currently targets PICO_BOARD=pico (RP2040)")
endif ()

add_executable(mock_hid main.c protocol.c usb_descriptors.c)
target_include_directories(mock_hid PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(mock_hid PRIVATE -Wall -Wextra -Werror)
target_link_libraries(mock_hid PRIVATE pico_stdlib hardware_uart tinyusb_device)
pico_enable_stdio_uart(mock_hid 0)
pico_enable_stdio_usb(mock_hid 0)
pico_add_extra_outputs(mock_hid)
